**Goal:** 在计划 1 的地基上，做出一个**真好玩、真想截图分享**的单机爽感闭环——许个愿 → 敲木鱼 Pro（核心玩具 + 汁水反馈 + 随机意外）→ 好运槽达成 → 生成「功德战报」+ 开一张「福气卡」→ 一键保存为图片/系统分享。**只验证一件事：有没有人觉得好玩到愿意玩、愿意分享。**

```ts
export enum WoodenFishEvent {
  NONE = 'none', CRIT = 'crit', SPIRIT = 'spirit', MIRACLE = 'miracle', BACKFIRE = 'backfire',
}
export function rollEvent({
  rng = Math.random, comboCount,
}: { rng?: () => number; comboCount: number }): WoodenFishEvent {
  const r = rng()
  if (r >= 0.995) return WoodenFishEvent.MIRACLE        // 0.5%
  if (r >= 0.97) return WoodenFishEvent.BACKFIRE         // ~2.5%
  if (r >= 0.85 && comboCount >= 20) return WoodenFishEvent.SPIRIT
  if (r >= 0.3) return WoodenFishEvent.CRIT
  return WoodenFishEvent.NONE
}
```

> 验证三处关键断言：`0.999→MIRACLE`、`0.5→CRIT`、`0.0→NONE`；改概率必须同步改测试。

```ts
import { FortuneCard, Rarity } from '@/types/game'
import { CARD_POOL } from './card-pool'

function pick(cards: FortuneCard[], rng: () => number): FortuneCard {
  const i = Math.min(Math.floor(rng() * cards.length), cards.length - 1)
  return cards[i]
}
export function drawCard({
  rng = Math.random, miracle,
}: { rng?: () => number; miracle: boolean }): FortuneCard {
  if (miracle) {
    return pick(CARD_POOL.filter((c) => c.rarity !== Rarity.COMMON), rng)
  }
  return pick(CARD_POOL, rng)
}
```

```ts
import { BattleReport } from '@/types/game'

// 玄学展示数：偏高、随敲击单调递增、封顶 99（非真实统计）
function vanityPercentile(taps: number): number {
  return Math.min(99, 70 + Math.floor(taps / 12))
}
function closerFor(percentile: number): string {
  if (percentile >= 95) return '佛祖已读，并亲自点了个赞 🙏'
  if (percentile >= 85) return '佛祖已收到，但建议你也去睡觉 😪'
  return '功德入账，记得也给自己充个电。'
}
export function buildReport({
  taps, luck, wish,
}: { taps: number; luck: number; wish?: string }): BattleReport {
  const percentile = vanityPercentile(taps)
  return { taps, luck, percentile, closer: closerFor(percentile), wish }
}
```
